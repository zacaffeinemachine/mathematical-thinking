export const Ring = ({ people }) => {
  const n = people.length, R = n === 3 ? 104 : 118, cx = 170, cy = n === 3 ? 142 : 156;
  const pos = people.map((_, k) => {
    const t = -Math.PI / 2 + (2 * Math.PI * k) / n;
    return [cx + R * Math.cos(t), cy + R * Math.sin(t)];
  });
  return (
    <svg className="isles-ring" viewBox={`0 0 340 ${n === 3 ? 278 : 330}`} width="340" role="img" aria-label={`${n} islanders in a circle, each speaking about the next`}>
      <defs>
        <marker id={`isles-arrow-${n}`} viewBox="0 0 10 10" refX="8" refY="5" markerWidth="8" markerHeight="8" orient="auto">
          <path d="M0 0 L10 5 L0 10 z" fill="var(--accent)" />
        </marker>
      </defs>
      {pos.map(([x1, y1], k) => {
        const [x2, y2] = pos[(k + 1) % n];
        const mx = (x1 + x2) / 2, my = (y1 + y2) / 2;
        const ml = Math.hypot(mx - cx, my - cy), ox = (mx - cx) / ml, oy = (my - cy) / ml;
        const c1 = [x1, y1 - 6], c2 = [x2, y2 - 6];
        const sx = c1[0] + (c2[0] - c1[0]) * 0.28, sy = c1[1] + (c2[1] - c1[1]) * 0.28;
        const ex = c1[0] + (c2[0] - c1[0]) * 0.72, ey = c1[1] + (c2[1] - c1[1]) * 0.72;
        return (
          <g>
            <path d={`M${sx} ${sy} Q${mx + ox * 16} ${my - 6 + oy * 16} ${ex} ${ey}`} fill="none" stroke="var(--accent)" stroke-width="1.8" marker-end={`url(#isles-arrow-${n})`} />
            <text x={mx + ox * 36} y={my - 6 + oy * 36 + 4} font-size="12" text-anchor="middle" fill="var(--accent)" font-style="italic">chimera</text>
          </g>
        );
      })}
      {pos.map(([x, y], k) => (
        <g>
          <Figure x={x} y={y - 16} robe={ROBES[k % ROBES.length]} s={0.8} />
          <text x={x} y={y + 31} font-size="12.5" font-weight="700" text-anchor="middle" fill="var(--ink)">{people[k].name}</text>
          <text x={x} y={y + 45} font-size="11" text-anchor="middle" fill="var(--muted)">switch <tspan font-family="ui-monospace, monospace" fill="var(--ink)">{people[k].sw}</tspan></text>
        </g>
      ))}
    </svg>
  );
};

export const Radios = ({ names }) => {
  const n = names.length;
  const pos = n === 3 ? [[150, 50], [240, 190], [60, 190]] : [[70, 50], [230, 50], [230, 190], [70, 190]];
  return (
    <svg className="isles-ring" viewBox="0 0 300 240" width="300" role="img" aria-label={`${n} radios, a report for each pair next to each other`}>
      {pos.map(([x1, y1], k) => {
        const [x2, y2] = pos[(k + 1) % n];
        const lab = names[k] + names[(k + 1) % n];
        const mx = (x1 + x2) / 2, my = (y1 + y2) / 2;
        return (
          <g>
            <line x1={x1} y1={y1} x2={x2} y2={y2} stroke="var(--muted)" stroke-width="1.6" stroke-dasharray="5 4" />
            <rect x={mx - 19} y={my - 11} width="38" height="22" rx="5" fill="var(--surface)" stroke="var(--muted)" />
            <text x={mx} y={my + 5} font-size="13" font-weight="700" text-anchor="middle" fill="var(--ink)" font-family="ui-monospace, monospace">{lab}</text>
          </g>
        );
      })}
      {pos.map(([x, y], k) => <Radio x={x} y={y} name={names[k]} />)}
    </svg>
  );
};
