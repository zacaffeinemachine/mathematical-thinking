import { Problem, Tests } from "../../components/games/fsm/Problem.tsx";

<Problem n={1} title="A warm-up">
Design a machine that accepts exactly the strings that end in `1`.
<Tests accept={["1", "01", "111", "0101"]} reject={["", "0", "10", "1110"]} />
</Problem>

<Problem n={2} title="Two at the end">
Design a machine that accepts exactly the strings that end in `00`.
<Tests accept={["00", "100", "0000", "11100"]} reject={["", "0", "01", "010", "0001"]} />
</Problem>

<Problem n={3} title="Three at the end">
Design a machine that accepts exactly the strings that end in `010`.
<Tests accept={["010", "0010", "1010", "01010"]} reject={["01", "0100", "10", "0110"]} />
</Problem>

<Problem n={4} title="One at the start">
Design a machine that accepts exactly the strings that begin with `1`.
<Tests accept={["1", "10", "1111", "1001"]} reject={["", "0", "01", "0111"]} />
</Problem>

<Problem n={5} title="Two at the start">
Design a machine that accepts exactly the strings that begin with `00`.
<Tests accept={["00", "001", "0011", "0000"]} reject={["", "0", "01", "10", "0101"]} />
</Problem>

<Problem n={6} title="Never leave a 0 hanging">
Design a machine that accepts exactly the strings in which every `0` is
immediately followed by a `1`.
<Tests accept={["", "1", "01", "1101", "010101"]} reject={["0", "10", "1001", "0110"]} />
</Problem>

<Problem n={7} title="Never let a 1 arrive alone">
Design a machine that accepts exactly the strings in which every `1` is
immediately preceded by a `0`.
<Tests accept={["", "0", "01", "0101", "0010"]} reject={["1", "10", "011", "0011"]} />
</Problem>

<Problem n={8} title="Zeros first">
Design a machine that accepts exactly the strings in which every `0` comes before every `1`.
<Tests accept={["", "0", "1", "000", "111", "00111"]} reject={["10", "010", "0101", "1101"]} />
</Problem>

<Problem n={9} title="The same at both ends" hard>
Design a machine that accepts exactly the strings of length at least 1 whose
first symbol and last symbol are the same.
<Tests accept={["0", "1", "00", "010", "1001", "11"]} reject={["", "01", "10", "0111", "1000"]} />
</Problem>

<Problem n={10} title="Both ends fixed" hard>
Design a machine that accepts exactly the strings that begin with `01` and end
with `10`.
<Tests accept={["010", "0110", "01010", "011110"]} reject={["", "01", "0101", "1010", "0011"]} />
</Problem>

<Problem n={11} title="A rule about the start that is not broken at the start" hard>
Design a machine that accepts exactly the strings whose first symbol appears
again somewhere later in the string.
<Tests accept={["00", "11", "0110", "1001", "010"]} reject={["", "0", "1", "01", "0111", "10"]} />
</Problem>
