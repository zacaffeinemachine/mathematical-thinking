import EquivalenceProblem from "../../components/games/fsm/EquivalenceProblem.tsx";
import {
  parityPlain,
  parityPadded,
  substring101,
  subsequence101,
  lastTwoSymbols,
  lastSymbolAndLength,
  containsEleven,
  someEvenBlockOfOnes,
  twoElevens,
  containsOneOneOne,
} from "../../components/games/fsm/equivalence.ts";

<EquivalenceProblem
  client:visible
  n={1}
  title="Two states against four"
  machineA={parityPlain}
  machineB={parityPadded}
  defaultInput="0110"
  examples={["", "1", "11", "0110", "1010"]}
>
Machine B has twice as many states as machine A. Do the two have the same
language, meaning that every string accepted by one is accepted by the other?
</EquivalenceProblem>

<EquivalenceProblem
  client:visible
  n={2}
  title="One arrow apart"
  machineA={substring101}
  machineB={subsequence101}
  defaultInput="1101"
  examples={["", "101", "1101", "0110"]}
>
These two drawings are identical except in one place, where a single arrow
goes somewhere else. Do they have the same language?
</EquivalenceProblem>

<EquivalenceProblem
  client:visible
  n={3}
  title="Loops against no loops"
  machineA={lastTwoSymbols}
  machineB={lastSymbolAndLength}
  defaultInput="0011"
  examples={["", "11", "0011", "0101"]}
>
Both machines have four states and both accept in two of them. Machine A has
two arrows that loop straight back to the state they left, and machine B has
none at all. Do they have the same language?
</EquivalenceProblem>

<EquivalenceProblem
  client:visible
  n={4}
  title="A state with no way out"
  machineA={containsEleven}
  machineB={someEvenBlockOfOnes}
  defaultInput="110"
  examples={["", "11", "110", "0110"]}
>
Each of these machines has a state that, once entered, it can never leave. In
machine A that state accepts and in machine B it accepts too, so neither
machine can ever change its mind once it arrives there. Do the two have the
same language?
</EquivalenceProblem>

<EquivalenceProblem
  client:visible
  n={5}
  title="Too long to stumble on"
  hard
  machineA={twoElevens}
  machineB={containsOneOneOne}
  defaultInput="1111"
  examples={["", "11", "111", "1111"]}
>
Every string of length four or less is accepted by both of these machines or
by neither, so typing short strings and watching what happens will tell you
nothing at all. Do the two have the same language?
</EquivalenceProblem>
