import { Problem, Tests } from "../../components/games/fsm/Problem.tsx";

<Problem n={1} title="Two in a row">
Design a machine that accepts exactly the strings that contain `00` somewhere inside.
<Tests accept={["00", "100", "0011", "10010"]} reject={["", "0", "1", "0101", "1010"]} />
</Problem>

<Problem n={2} title="Three in a row">
Design a machine that accepts exactly the strings that contain `101` somewhere inside.
<Tests accept={["101", "0101", "1011", "11010"]} reject={["", "10", "100", "0110", "1100"]} />
</Problem>

<Problem n={3} title="Four in a row">
Design a machine that accepts exactly the strings that contain `0110`
somewhere inside.

This is where the temptation to reset all the way to the beginning does the
most damage. Look hard at what happens when the machine has seen `011` and
another `1` arrives.
<Tests accept={["0110", "10110", "01100", "001101"]} reject={["", "011", "0101", "0111", "1100"]} />
</Problem>

<Problem n={4} title="Never in a row">
Design a machine that accepts exactly the strings that do **not** contain `11`
anywhere.

You can build this from the machine for *contains `11`* without redrawing a
single arrow. Say what you did, and why it is allowed.
<Tests accept={["", "0", "1", "01", "1010", "0100"]} reject={["11", "011", "110", "1011"]} />
</Problem>

<Problem n={5} title="Both of two blocks">
Design a machine that accepts exactly the strings that contain `01` somewhere **and** contain `10`
somewhere.
<Tests accept={["010", "101", "0110", "1001"]} reject={["", "01", "10", "0011", "11"]} />
</Problem>

<Problem n={6} title="A word in the middle of a sentence" alphabet="{a, b, c}">
Design a machine that accepts exactly the strings that contain `abc`
somewhere inside.
<Tests accept={["abc", "aabc", "abcc", "cabca"]} reject={["", "ab", "acb", "abab", "bca"]} />
</Problem>

<Problem n={7} title="One block only if the other one turns up" hard>
Design a machine that accepts exactly the strings that contain `00` **if**
they contain `11`.

A string with no `11` in it satisfies the rule for free, whatever else it
does. A string with a `11` in it has to have a `00` somewhere too, and the
`00` may come before or after.
<Tests accept={["", "01", "0100", "0011", "1100", "1010"]} reject={["11", "011", "110", "1101"]} />
</Problem>

<Problem n={8} title="Once and once only" hard>
Design a machine that accepts exactly the strings that contain `11` exactly
once.

Overlapping counts as more than once, so `111` contains `11` twice: once at
the first two symbols and once at the last two. Decide what your machine
should do with `111` before you draw it.
<Tests accept={["11", "011", "110", "0110", "11010"]} reject={["", "1", "111", "1111", "11011"]} />
</Problem>

<Problem n={9} title="A block in every other symbol" hard alphabet="the 26 lowercase letters">
The **odd part** of a string is what is left when every symbol at an even
position is deleted, so the odd part of `dxfxa` is `dfa`, and the odd part of
`ihateqrmt` is `iaert`.

Design a machine that accepts exactly the strings whose odd part contains
`dfa` somewhere inside. Draw one arrow labelled `x` for "any other letter"
rather than drawing twenty-six.

Note that `dfa` itself is not accepted. Work out its odd part and you will
see why.
<Tests accept={["dxfxa", "dafaa", "zzdxfxa", "dqfsa"]} reject={["", "dfa", "dxf", "xdxfxa", "adf"]} />
</Problem>
