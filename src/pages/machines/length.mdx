import { Problem, Tests } from "../../components/games/fsm/Problem.tsx";

<Problem n={1} title="Exactly three">
Design a machine that accepts exactly the strings of length 3.

Watch the strings longer than 3. They are not accepted, and they are never
going to be, so where do they go?
<Tests accept={["000", "101", "111", "010"]} reject={["", "1", "10", "0000", "10101"]} />
</Problem>

<Problem n={2} title="At most two">
Design a machine that accepts exactly the strings of length 2 or less.
<Tests accept={["", "0", "1", "00", "10", "11"]} reject={["000", "101", "1010", "111"]} />
</Problem>

<Problem n={3} title="At least four">
Design a machine that accepts exactly the strings of length 4 or more.
<Tests accept={["0000", "1010", "00000", "1111011"]} reject={["", "0", "01", "010", "111"]} />
</Problem>

<Problem n={4} title="Even length">
Design a machine that accepts exactly the strings of even length. The empty string has length 0,
which is even.
<Tests accept={["", "01", "10", "0000", "1101"]} reject={["0", "111", "10101", "1"]} />
</Problem>

<Problem n={5} title="A multiple of three">
Design a machine that accepts exactly the strings whose length is a multiple of 3.
<Tests accept={["", "010", "111", "101010", "000000"]} reject={["0", "01", "1010", "11111"]} />
</Problem>

<Problem n={6} title="Length and last symbol together">
Design a machine that accepts exactly the strings whose length is a multiple
of 3 **and** whose last symbol is `1`.

Two things to remember at once. Notice that they are not independent: once
you know the length is a multiple of 3, the last symbol still has to be
checked, and the machine has to have been checking it all along.
<Tests accept={["001", "111", "010101", "101011"]} reject={["", "1", "11", "010", "0101"]} />
</Problem>

<Problem n={7} title="Even or a multiple of three" hard>
Design a machine that accepts exactly the strings whose length is even **or**
a multiple of 3, or both.

The accepted lengths are 0, 2, 3, 4, 6, 8, 9, 10, and so on. The rejected
ones are 1, 5, 7, 11, 13. How many circles does the ring need?
<Tests accept={["", "01", "010", "0101", "010101", "01010101"]} reject={["0", "01010", "0101010", "01010101010"]} />
</Problem>

<Problem n={8} title="A symbol at a fixed position" hard>
Design a machine that accepts exactly the strings that have length at least 3 and whose **third**
symbol is a `0`.
<Tests accept={["000", "110", "1101", "0000"]} reject={["", "1", "10", "111", "1011"]} />
</Problem>

<Problem n={9} title="Every other position" hard>
Design a machine that accepts exactly the strings in which every symbol at an odd-numbered position
is a `1`. Positions are counted from 1, so this is the 1st symbol, the 3rd,
the 5th, and so on. The symbols at even positions may be anything.
<Tests accept={["", "1", "10", "11", "1010", "1011"]} reject={["0", "01", "0101", "1100", "110"]} />
</Problem>

<Problem n={10} title="The first symbol decides which rule applies" hard>
Design a machine that accepts exactly the strings that either begin with `1`
and have even length, or begin with `0` and have odd length.

The empty string begins with neither, so it is rejected.
<Tests accept={["0", "10", "010", "1010", "00000"]} reject={["", "1", "00", "0101", "101"]} />
</Problem>
